from collections import deque

def water_jug():
    queue = deque([(0, 0)])
    visited = set()

    while queue:
        a, b = queue.popleft()

        if (a, b) in visited:
            continue
        visited.add((a, b))

        print(a, b)

        if a == 2 or b == 2:
            print("Goal Reached!")
            return

        states = [
            (4, b), (a, 3), (0, b), (a, 0),
            (a - min(a, 3-b), b + min(a, 3-b)),
            (a + min(b, 4-a), b - min(b, 4-a))
        ]

        for state in states:
            if state not in visited:
                queue.append(state)

water_jug()

# Game Tree Representation for Two-Player Game

tree = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F', 'G'],
    'D': [3, 5],
    'E': [6, 9],
    'F': [1, 2],
    'G': [0, -1]
}

def minimax(node, maximizing):
    if isinstance(tree[node][0], int):
        return max(tree[node]) if maximizing else min(tree[node])

    if maximizing:
        value = float('-inf')
        for child in tree[node]:
            value = max(value, minimax(child, False))
        return value
    else:
        value = float('inf')
        for child in tree[node]:
            value = min(value, minimax(child, True))
        return value

print("Game Tree:", tree)
print("Best Value:", minimax('A', True))
