import heapq

graph = {
    'A': [('B', 1), ('C', 4)],
    'B': [('D', 2), ('E', 5)],
    'C': [('E', 1)],
    'D': [('F', 3)],
    'E': [('F', 2)],
    'F': []
}

h = {'A': 6, 'B': 5, 'C': 3, 'D': 3, 'E': 2, 'F': 0}

def astar(start, goal):
    pq = [(h[start], 0, start, [start])]
    best = {start: 0}

    while pq:
        _, g, node, path = heapq.heappop(pq)

        if node == goal:
            return path, g

        if g != best[node]:
            continue

        for nxt, w in graph[node]:
            new_g = g + w

            if new_g < best.get(nxt, float('inf')):
                best[nxt] = new_g
                heapq.heappush(pq, (new_g + h[nxt], new_g, nxt, path + [nxt]))

    return None, float('inf')

path, cost = astar('A', 'F')

print("Path:", path)
print("Cost:", cost)
