import heapq

graph = {
    'A': [('B', 2), ('C', 4)],
    'B': [('A', 2), ('D', 3), ('E', 5)],
    'C': [('A', 4), ('F', 1)],
    'D': [('B', 3), ('F', 2)],
    'E': [('B', 5), ('F', 1)],
    'F': [('C', 1), ('D', 2), ('E', 1)]
}

h = {'A': 6, 'B': 5, 'C': 3, 'D': 2, 'E': 1, 'F': 0}

def best_first(start, goal):
    pq = [(h[start], start, [start])]
    visited = set()

    while pq:
        _, node, path = heapq.heappop(pq)

        if node == goal:
            return path

        if node in visited:
            continue

        visited.add(node)

        for neighbor, _ in graph[node]:
            if neighbor not in visited:
                heapq.heappush(pq, (h[neighbor], neighbor, path + [neighbor]))

    return None

print("Path:", best_first('A', 'F'))

# Propositional Logic using AND, OR and IMPLIES



p = True
q = False

AND = p and q
OR = p or q
IMPLIES = (not p) or q

print("P =", p)
print("Q =", q)
print("P AND Q =", AND)
print("P OR Q =", OR)
print("P IMPLIES Q =", IMPLIES)
